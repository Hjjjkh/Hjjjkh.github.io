Spring Boot项目实战：构建RESTful API

本文最后更新于 2025年11月15日 下午

项目背景

本文将介绍如何使用Spring Boot快速构建一个完整的RESTful API项目，包括用户管理、数据验证、异常处理等核心功能。

技术栈

  • Spring Boot 3.x
  • Spring Data JPA
  • MySQL
  • Maven
  • Lombok

项目结构

src/main/java/com/example/demo
├── controller      # 控制器层
├── service         # 业务逻辑层
├── repository      # 数据访问层
├── entity          # 实体类
├── dto             # 数据传输对象
├── exception       # 异常处理
└── config          # 配置类

核心代码实现

1. 实体类定义

@Entity
@Table(name = "users")
@Data
@NoArgsConstructor
@AllArgsConstructor
public class User &#123;
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    
    @Column(nullable = false, unique = true)
    private String username;
    
    @Column(nullable = false)
    private String email;
    
    @Column(nullable = false)
    private String password;
    
    @CreatedDate
    private LocalDateTime createdAt;
    
    @LastModifiedDate
    private LocalDateTime updatedAt;

public interface UserRepository extends JpaRepository<User, Long> &#123;

3. Service层

@Service
@RequiredArgsConstructor
public class UserService &#123;
    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;
    
    public UserDTO createUser(CreateUserRequest request) &#123;
        // 验证用户名是否已存在
        if (userRepository.existsByUsername(request.getUsername())) &#123;
            throw new DuplicateResourceException("用户名已存在");
        &#125;
        
        // 验证邮箱是否已存在
        if (userRepository.existsByEmail(request.getEmail())) &#123;
            throw new DuplicateResourceException("邮箱已被注册");
        &#125;
        
        // 创建用户
        User user = new User();
        user.setUsername(request.getUsername());
        user.setEmail(request.getEmail());
        user.setPassword(passwordEncoder.encode(request.getPassword()));
        
        User savedUser = userRepository.save(user);
        return convertToDTO(savedUser);
    &#125;
    
    public UserDTO getUserById(Long id) &#123;
        User user = userRepository.findById(id)
            .orElseThrow(() -> new ResourceNotFoundException("用户不存在"));
        return convertToDTO(user);
    &#125;
    
    public List<UserDTO> getAllUsers() &#123;
        return userRepository.findAll().stream()
            .map(this::convertToDTO)
            .collect(Collectors.toList());
    &#125;
    
    public UserDTO updateUser(Long id, UpdateUserRequest request) &#123;
        User user = userRepository.findById(id)
            .orElseThrow(() -> new ResourceNotFoundException("用户不存在"));
        
        if (request.getEmail() != null) &#123;
            user.setEmail(request.getEmail());
        &#125;
        
        User updatedUser = userRepository.save(user);
        return convertToDTO(updatedUser);
    &#125;
    
    public void deleteUser(Long id) &#123;
        if (!userRepository.existsById(id)) &#123;
            throw new ResourceNotFoundException("用户不存在");
        &#125;
        userRepository.deleteById(id);
    &#125;
    
    private UserDTO convertToDTO(User user) &#123;
        return new UserDTO(
            user.getId(),
            user.getUsername(),
            user.getEmail(),
            user.getCreatedAt()
        );
    &#125;

4. Controller层

@RestController
@RequestMapping("/api/users")
@RequiredArgsConstructor
@Validated
public class UserController &#123;
    private final UserService userService;
    
    @PostMapping
    public ResponseEntity<ApiResponse<UserDTO>> createUser(
            @Valid @RequestBody CreateUserRequest request) &#123;
        UserDTO user = userService.createUser(request);
        return ResponseEntity.status(HttpStatus.CREATED)
            .body(ApiResponse.success(user, "用户创建成功"));
    &#125;
    
    @GetMapping("/&#123;id&#125;")
    public ResponseEntity<ApiResponse<UserDTO>> getUserById(@PathVariable Long id) &#123;
        UserDTO user = userService.getUserById(id);
        return ResponseEntity.ok(ApiResponse.success(user));
    &#125;
    
    @GetMapping
    public ResponseEntity<ApiResponse<List<UserDTO>>> getAllUsers() &#123;
        List<UserDTO> users = userService.getAllUsers();
        return ResponseEntity.ok(ApiResponse.success(users));
    &#125;
    
    @PutMapping("/&#123;id&#125;")
    public ResponseEntity<ApiResponse<UserDTO>> updateUser(
            @PathVariable Long id,
            @Valid @RequestBody UpdateUserRequest request) &#123;
        UserDTO user = userService.updateUser(id, request);
        return ResponseEntity.ok(ApiResponse.success(user, "用户更新成功"));
    &#125;
    
    @DeleteMapping("/&#123;id&#125;")
    public ResponseEntity<ApiResponse<Void>> deleteUser(@PathVariable Long id) &#123;
        userService.deleteUser(id);
        return ResponseEntity.ok(ApiResponse.success(null, "用户删除成功"));
    &#125;
&#125;

5. 全局异常处理

@RestControllerAdvice
public class GlobalExceptionHandler &#123;
    
    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ApiResponse<Void>> handleResourceNotFound(
            ResourceNotFoundException ex) &#123;
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
            .body(ApiResponse.error(ex.getMessage()));
    &#125;
    
    @ExceptionHandler(DuplicateResourceException.class)
    public ResponseEntity<ApiResponse<Void>> handleDuplicateResource(
            DuplicateResourceException ex) &#123;
        return ResponseEntity.status(HttpStatus.CONFLICT)
            .body(ApiResponse.error(ex.getMessage()));
    &#125;
    
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Map<String, String>>> handleValidationErrors(
            MethodArgumentNotValidException ex) &#123;
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getFieldErrors().forEach(error ->
            errors.put(error.getField(), error.getDefaultMessage())
        );
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
            .body(ApiResponse.error("数据验证失败", errors));
    &#125;
    
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiResponse<Void>> handleGenericException(Exception ex) &#123;
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(ApiResponse.error("服务器内部错误"));
    &#125;
&#125;

6. 统一响应格式

@Data
@AllArgsConstructor
public class ApiResponse<T> &#123;
    private boolean success;
    private String message;
    private T data;
    private LocalDateTime timestamp;
    
    public static <T> ApiResponse<T> success(T data) &#123;
        return new ApiResponse<>(true, "操作成功", data, LocalDateTime.now());
    &#125;
    
    public static <T> ApiResponse<T> success(T data, String message) &#123;
        return new ApiResponse<>(true, message, data, LocalDateTime.now());
    &#125;
    
    public static <T> ApiResponse<T> error(String message) &#123;
        return new ApiResponse<>(false, message, null, LocalDateTime.now());
    &#125;
    
    public static <T> ApiResponse<T> error(String message, T data) &#123;
        return new ApiResponse<>(false, message, data, LocalDateTime.now());
    &#125;

配置文件

application.yml

spring:
  application:
    name: demo-api
  
  datasource:
    url: jdbc:mysql://localhost:3306/demo_db?useUnicode=true&characterEncoding=utf8
    username: root
    password: your_password
    driver-class-name: com.mysql.cj.jdbc.Driver
  
  jpa:
    hibernate:
      ddl-auto: update
    show-sql: true
    properties:
      hibernate:
        format_sql: true
        dialect: org.hibernate.dialect.MySQL8Dialect
  
server:
  port: 8080
  servlet:
    context-path: /

logging:
  level:
    root: INFO
    com.example.demo: DEBUG

项目亮点

  1. 分层架构清晰：Controller、Service、Repository职责明确
  2. 统一异常处理：使用@RestControllerAdvice集中处理异常
  3. 数据验证：使用Bean Validation进行参数校验
  4. 统一响应格式：所有API返回统一的数据结构
  5. RESTful设计：遵循REST API设计规范
